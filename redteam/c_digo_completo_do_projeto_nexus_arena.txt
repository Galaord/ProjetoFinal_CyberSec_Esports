from django.db import models

class Equipa(models.Model):
    nome = models.CharField(max_length=100)
    tag = models.CharField(max_length=10)
    logo_url = models.URLField(blank=True, null=True)
    pais = models.CharField(max_length=50, default="Portugal")
    vitorias = models.IntegerField(default=0)
    derrotas = models.IntegerField(default=0)

    class Meta:
        db_table = 'equipas'

    def __str__(self):
        return f"[{self.tag}] {self.nome}"


class Utilizador(models.Model):
    nickname = models.CharField(max_length=50, unique=True)
    email = models.EmailField(unique=True)
    palavra_passe = models.CharField(max_length=255)
    steam_id = models.CharField(max_length=50, blank=True, null=True)
    elo = models.IntegerField(default=1000)
    equipa = models.ForeignKey(Equipa, on_delete=models.SET_NULL, null=True, blank=True, related_name="jogadores")
    data_registo = models.DateTimeField(auto_now_add=True)

    class Meta:
        db_table = 'utilizadores'

    def __str__(self):
        return self.nickname


class Torneio(models.Model):
    nome = models.CharField(max_length=100)
    jogo = models.CharField(max_length=50, default="Counter-Strike 2")
    formato = models.CharField(max_length=50, default="5v5")
    premio = models.CharField(max_length=50, default="1.000€")
    estado = models.CharField(max_length=20, default="Planeado") # ex: Planeado, Em andamento, Concluído
    data_inicio = models.DateField()
    data_fim = models.DateField(null=True, blank=True)

    class Meta:
        db_table = 'torneios'

    def __str__(self):
        return self.nome


class Jogo(models.Model):
    torneio = models.ForeignKey(Torneio, on_delete=models.CASCADE, related_name="jogos")
    equipa_a = models.ForeignKey(Equipa, on_delete=models.CASCADE, related_name="jogos_como_a")
    equipa_b = models.ForeignKey(Equipa, on_delete=models.CASCADE, related_name="jogos_como_b")
    resultado_a = models.IntegerField(default=0)
    resultado_b = models.IntegerField(default=0)
    data_hora = models.DateTimeField()
    mapa = models.CharField(max_length=50, default="Mirage")
    transmissao_url = models.URLField(blank=True, null=True)

    class Meta:
        db_table = 'jogos'

    def __str__(self):
        return f"{self.equipa_a.tag} vs {self.equipa_b.tag} ({self.torneio.nome})"

from django.shortcuts import render, redirect, get_object_or_404
from django.contrib import messages
from .models import Equipa, Utilizador, Torneio, Jogo

def pagina_inicial(request):
    proximos_jogos = Jogo.objects.all().order_by('data_hora')[:3]
    ultimos_torneios = Torneio.objects.all().order_by('-data_inicio')[:3]
    return render(request, 'PaginaInicial_2.html', {
        'jogos': proximos_jogos,
        'torneios': ultimos_torneios
    })

def jogos(request):
    lista_jogos = Jogo.objects.all().order_by('data_hora')
    return render(request, 'jogos.html', {'jogos': lista_jogos})

def torneios(request):
    lista_torneios = Torneio.objects.all().order_by('-data_inicio')
    return render(request, 'torneios_2.html', {'torneios': lista_torneios})

def detalhe_torneio(request, torneio_id):
    torneio = get_object_or_404(Torneio, id=torneio_id)
    jogos_do_torneio = Jogo.objects.filter(torneio=torneio).order_by('data_hora')
    return render(request, 'detalhe_torneio.html', {
        'torneio': torneio,
        'jogos': jogos_do_torneio
    })

def comunidade(request):
    top_jogadores = Utilizador.objects.all().order_by('-elo')[:10]
    equipas = Equipa.objects.all()
    return render(request, 'Comunidade_2.html', {
        'jogadores': top_jogadores,
        'equipas': equipas
    })

def login_view(request):
    if request.method == 'POST':
        nickname = request.POST.get('nickname')
        password = request.POST.get('password')
        try:
            user = Utilizador.objects.get(nickname=nickname, palavra_passe=password)
            request.session['utilizador_id'] = user.id
            request.session['utilizador_nickname'] = user.nickname
            messages.success(request, f"Bem-vindo de volta, {user.nickname}!")
            return redirect('pagina_inicial')
        except Utilizador.DoesNotExist:
            messages.error(request, "Credenciais inválidas. Tenta novamente.")
    return render(request, 'login_2.html')

def registo_view(request):
    if request.method == 'POST':
        nickname = request.POST.get('nickname')
        email = request.POST.get('email')
        password = request.POST.get('password')
        steam_id = request.POST.get('steam_id', '')

        if Utilizador.objects.filter(nickname=nickname).exists():
            messages.error(request, "Este Nickname já está em uso.")
            return render(request, 'registo_2.html')

        if Utilizador.objects.filter(email=email).exists():
            messages.error(request, "Este Email já está registado.")
            return render(request, 'registo_2.html')

        Utilizador.objects.create(
            nickname=nickname,
            email=email,
            palavra_passe=password,
            steam_id=steam_id
        )
        messages.success(request, "Conta criada com sucesso! Podes agora iniciar sessão.")
        return redirect('login')

    return render(request, 'registo_2.html')

def logout_view(request):
    request.session.flush()
    messages.info(request, "Sessão terminada.")
    return redirect('pagina_inicial')

def teste_bd(request):
    equipas = Equipa.objects.all()
    return render(request, 'teste.html', {'equipas': equipas})
